
import java.util.*;

public class Main {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of items: ");
        int n = sc.nextInt();

        int[] weight = new int[n];
        int[] value = new int[n];

        for (int i = 0; i < n; i++) {
            System.out.print("Enter weight and value: ");
            weight[i] = sc.nextInt();
            value[i] = sc.nextInt();
        }

        System.out.print("Enter capacity: ");
        int W = sc.nextInt();

        int[][] dp = new int[n + 1][W + 1];

    
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= W; j++) {

                if (weight[i - 1] <= j) {
                    dp[i][j] = Math.max(
                        dp[i - 1][j],
                        value[i - 1] + dp[i - 1][j - weight[i - 1]]
                    );
                } else {
                    dp[i][j] = dp[i - 1][j];
                }
            }
        }

        System.out.println("Maximum Profit = " + dp[n][W]);

       
        int j = W;

        System.out.print("Selected Items: ");

        for (int i = n; i > 0; i--) {
            if (dp[i][j] != dp[i - 1][j]) {
                System.out.print(i + " ");
                j = j - weight[i - 1];
            }
        }

        sc.close();
    }
}
